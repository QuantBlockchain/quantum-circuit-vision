[0m[38;5;10mfrom braket.circuits import Circuit
import math

circ = Circuit()

# Step 1: Hadamard on even qubits
circ.h([0, 2, 4, 6])

# Step 2: First layer of CNOTs (even -> odd)
circ.cnot(0, 1)
circ.cnot(2, 3)
circ.cnot(4, 5)
circ.cnot(6, 7)

# Step 3: Second layer of CNOTs (odd -> next even, chaining)
circ.cnot(1, 2)
circ.cnot(3, 4)
circ.cnot(5, 6)

# Step 4: Long-range CNOT (q7 -> q0)
circ.cnot(7, 0)

# Step 5: Rz(pi/4) on even qubits
circ.rz(0, math.pi / 4)
circ.rz(2, math.pi / 4)
circ.rz(4, math.pi / 4)
circ.rz(6, math.pi / 4)

# Step 6: Hadamard on even qubits
circ.h([0, 2, 4, 6])

print(circ)
