[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

c = Circuit()

# --- Initial Hadamard layer ---
c.h([0, 1, 2, 3])

# === QAOA Layer 1 ===

# Edge (q0, q1)
c.cnot(0, 1).rz(1, np.pi/2).cnot(0, 1)

# Edge (q1, q2)
c.cnot(1, 2).rz(2, np.pi/2).cnot(1, 2)

# Edge (q2, q3)
c.cnot(2, 3).rz(3, np.pi/2).cnot(2, 3)

# Mixer layer 1
c.rx(1, np.pi/4)

# Edge (q2, q0) long-range
c.cnot(2, 0)

c.rx(2, np.pi/4)

# Edge (q3, q0) long-range
c.cnot(3, 0).rz(0, np.pi/2).cnot(3, 0)

c.rx(0, np.pi/4)
c.rx(3, np.pi/4)

# === QAOA Layer 2 ===

# Edge (q0, q1)
c.cnot(0, 1).rz(1, np.pi/3).cnot(0, 1)

# Edge (q1, q2)
c.cnot(1, 2).rz(2, np.pi/3).cnot(1, 2)

# Edge (q2, q3)
c.cnot(2, 3).rz(3, np.pi/3).cnot(2, 3)

# Mixer layer 2
c.rx(1, np.pi/5)

# Edge (q2, q0) long-range
c.cnot(2, 0)

# Edge (q3, q0) long-range
c.cnot(3, 0).rz(0, np.pi/3).cnot(3, 0)

c.rx(2, np.pi/5)
c.rx(3, np.pi/5)

# Tail: final CNOT(q3->q0) + Rx mixer
c.cnot(3, 0)
c.rx(0, np.pi/5)
c.rx(3, np.pi/5)

print(c)
