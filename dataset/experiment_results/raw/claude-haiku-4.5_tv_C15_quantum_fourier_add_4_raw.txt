import numpy as np

circuit = Circuit()

# q0: H gate
circuit.h(0)

# Controlled-P(π/2): q0 controls q1
circuit.cphaseshift(0, 1, np.pi / 2)

# q1: H gate
circuit.h(1)

# Controlled-P(π/2): q0 controls q2
circuit.cphaseshift(0, 2, np.pi / 2)

# Controlled-P(π/4): q2 controls q3
circuit.cphaseshift(2, 3, np.pi / 4)

# Controlled-P(π/2): q1 controls q3
circuit.cphaseshift(1, 3, np.pi / 2)

# q1: H gate
circuit.h(1)

# Controlled-P(-π/2): q0 controls q1
circuit.cphaseshift(0, 1, -np.pi / 2)

# q0: H gate
circuit.h(0)

print(circuit)
