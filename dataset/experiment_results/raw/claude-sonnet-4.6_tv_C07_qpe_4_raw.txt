[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circ = Circuit()

# 初始化
circ.h(0)
circ.h(1)
circ.x(3)

# 受控相位门（QPE 相位反冲部分）
circ.cphaseshift(0, 2, np.pi / 2)
circ.cphaseshift(0, 3, np.pi / 4)

# q0 上的 H（QFT 逆变换开始）
circ.h(0)

circ.cphaseshift(1, 2, np.pi / 4)
circ.cphaseshift(1, 3, np.pi / 8)

# 受控相位 + H（QFT 逆变换继续）
circ.cphaseshift(0, 1, -np.pi / 2)
circ.h(1)

# SWAP q0 ↔ q1
circ.swap(0, 1)

print(circ)
