[0m[38;5;10mfrom braket.circuits import Circuit

circuit = Circuit()

# First layer: Rx gates
circuit.rx(0, angle=3.14159/3)      # Rx(π/3) on q0
circuit.rx(1, angle=3.14159/4)      # Rx(π/4) on q1
circuit.rx(2, angle=3.14159/5)      # Rx(π/5) on q2
circuit.rx(3, angle=3.14159/6)      # Rx(π/6) on q3

# Second layer: Rz gates
circuit.rz(0, angle=3.14159/5)      # Rz(π/5) on q0
circuit.rz(1, angle=3.14159/5)      # Rz(π/5) on q1
circuit.rz(2, angle=3.14159/6)      # Rz(π/6) on q2
circuit.rz(3, angle=3.14159/7)      # Rz(π/7) on q3

# Third layer: CNOT gates
circuit.cnot(0, 1)                  # CNOT with control q0, target q1

# Fourth layer: Rz gate
circuit.rz(0, angle=3.14159/5)      # Rz(π/5) on q0

# Fifth layer: CNOT gates
circuit.cnot(1, 2)                  # CNOT with control q1, target q2

# Sixth layer: CNOT gate
circuit.cnot(2, 3)                  # CNOT with control q2, target q3

# Seventh layer: Rz gates
circuit.rz(1, angle=3.14159/6)      # Rz(π/6) on q1
circuit.rz(2, angle=3.14159/7)      # Rz(π/7) on q2
circuit.rz(3, angle=3.14159/8)      # Rz(π/8) on q3

print(circuit)
