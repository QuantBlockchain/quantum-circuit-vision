[0m[38;5;10mimport math
from braket.circuits import Circuit

circuit = Circuit()

# Ry(1.91) on q0
circuit.ry(0, 1.91)

# Controlled-H: control=q0, target=q1
# CH = Ry(pi/4) -> CNOT -> Ry(-pi/4)
circuit.ry(1, math.pi / 4)
circuit.cnot(0, 1)
circuit.ry(1, -math.pi / 4)

# CNOT: control=q1, target=q2
circuit.cnot(1, 2)

# CNOT: control=q0, target=q1
circuit.cnot(0, 1)

# X on q0
circuit.x(0)

print(circuit)
