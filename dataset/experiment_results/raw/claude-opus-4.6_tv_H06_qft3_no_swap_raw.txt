[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# Step 1: H on q0
circ.h(0)

# Step 2: Controlled-P(pi/2), control=q0, target=q1
circ.cphaseshift(0, 1, math.pi / 2)

# Step 3: Controlled-P(pi/4), control=q0, target=q2
circ.cphaseshift(0, 2, math.pi / 4)

# Step 4: H on q1
circ.h(1)

# Step 5: Controlled-P(pi/2), control=q1, target=q2
circ.cphaseshift(1, 2, math.pi / 2)

# Step 6: H on q2
circ.h(2)

print(circ)
