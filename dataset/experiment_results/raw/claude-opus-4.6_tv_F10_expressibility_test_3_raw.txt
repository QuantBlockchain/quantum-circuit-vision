[0m[38;5;10mimport math
from braket.circuits import Circuit

circuit = Circuit()

# Layer 1: Rx
circuit.rx(0, math.pi / 2)
circuit.rx(1, math.pi / 3)
circuit.rx(2, math.pi / 4)

# Layer 2: Rz
circuit.rz(0, math.pi / 3)
circuit.rz(1, math.pi / 4)
circuit.rz(2, math.pi / 5)

# Layer 3: CNOT entanglement
circuit.cnot(0, 1)
circuit.cnot(1, 2)
circuit.cnot(1, 0)
circuit.cnot(2, 0)

# Layer 4: Ry
circuit.ry(0, math.pi / 4)
circuit.ry(1, math.pi / 5)
circuit.ry(2, math.pi / 6)

# Layer 5: Rz
circuit.rz(0, math.pi / 5)
circuit.rz(1, math.pi / 6)
circuit.rz(2, math.pi / 7)

print(circuit)
