# Middle RY gate

# More CNOT gates

circuit.cnot(0, 3)

# Final RY gates
circuit.ry(0, math.pi / 5)
circuit.ry(2, math.pi / 7)
circuit.ry(3, math.pi / 8)

print(circuit)
