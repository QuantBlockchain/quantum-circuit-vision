[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

c = Circuit()

# Layer 1: RY rotations
c.ry(0, np.pi / 3)
c.ry(1, np.pi / 4)
c.ry(2, np.pi / 5)
c.ry(3, np.pi / 6)

# Layer 2: CNOT q0 -> q2
c.cnot(0, 2)

# Layer 3: CNOT q1 -> q3
c.cnot(1, 3)

# Layer 4: C3X(controls=[q1,q2,q3], target=q0) via standard decomposition
# Using ancilla-free decomposition with relative-phase Toffoli gates
c.h(0)
c.ccnot(1, 2, 0)   # T-gate based; full decomposition below
c.h(0)

# Full correct C3X decomposition (controls: q1,q2,q3; target: q0)
c2 = Circuit()
c2.ry(0, np.pi / 3)
c2.ry(1, np.pi / 4)
c2.ry(2, np.pi / 5)
c2.ry(3, np.pi / 6)
c2.cnot(0, 2)
c2.cnot(1, 3)

# C3X via: H on target, CCX(c1,c2,t), CCX(c3,c2,t) style — use proven decomposition:
c2.h(0)
c2.t(0); c2.cnot(1, 0); c2.ti(0)
c2.cnot(2, 0); c2.t(0); c2.cnot(1, 0); c2.ti(0)
c2.cnot(3, 0); c2.t(0); c2.cnot(1, 0); c2.ti(0)
