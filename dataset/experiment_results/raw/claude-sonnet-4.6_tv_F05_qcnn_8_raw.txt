circ = Circuit()

for q in range(8):
    circ.ry(q, np.pi / 4)

# 列2: CNOT pairs (control→target)

    circ.cnot(ctrl, tgt)

# 列3: CNOT pairs again
for ctrl, tgt in [(0, 1), (2, 3), (4, 5), (6, 7)]:
    circ.cnot(ctrl, tgt)

# 列4: RY(π/3) on q1, q3, q5, q7
for q in [1, 3, 5, 7]:
    circ.ry(q, np.pi / 3)

# 列5: CNOT (q1→q3), (q5→q7)
for ctrl, tgt in [(1, 3), (5, 7)]:
    circ.cnot(ctrl, tgt)

# 列6: CNOT (q1→q3), (q5→q7)

    circ.cnot(ctrl, tgt)
