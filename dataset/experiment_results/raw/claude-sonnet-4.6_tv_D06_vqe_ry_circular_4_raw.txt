[0m[38;5;10mimport math
from braket.circuits import Circuit

c = Circuit()

# Layer 1: RY rotations
c.ry(0, math.pi / 3)
c.ry(1, math.pi / 4)
c.ry(2, math.pi / 5)
c.ry(3, math.pi / 6)

# Layer 2: Circular CNOT entanglement
c.cnot(0, 1)
c.cnot(1, 2)
c.cnot(2, 3)
c.cnot(3, 0)

# Mid-layer RY (q1 after CNOT q0→q1, before CNOT q1→q2 resolves; placed after entanglement)
c.ry(1, math.pi / 6)

# Layer 3: Final RY rotations
c.ry(0, math.pi / 5)
c.ry(2, math.pi / 7)
c.ry(3, math.pi / 8)

print(c)
