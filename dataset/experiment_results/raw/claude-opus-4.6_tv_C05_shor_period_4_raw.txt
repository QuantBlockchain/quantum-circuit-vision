[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circuit = Circuit()

# 初始层
circuit.h(0)
circuit.h(1)
circuit.x(3)

# 受控门序列 (模幂运算)
circuit.cnot(1, 2)
circuit.ccnot(1, 2, 3)
circuit.cnot(0, 3)
circuit.cnot(2, 3)

circuit.h(0)
circuit.cphaseshift(1, 0, -np.pi / 2)
circuit.h(1)

print(circuit)
