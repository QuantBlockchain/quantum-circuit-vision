[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# QFT on q0
circ.h(0)
circ.cphaseshift(1, 0, math.pi / 2)
circ.cphaseshift(2, 0, math.pi / 4)
circ.cphaseshift(3, 0, math.pi / 8)

# QFT on q1
circ.h(1)
circ.cphaseshift(2, 1, math.pi / 2)
circ.cphaseshift(3, 1, math.pi / 4)

# QFT on q2
circ.h(2)
circ.cphaseshift(3, 2, math.pi / 2)

# QFT on q3
circ.h(3)

# Bit-reversal swaps
circ.swap(0, 3)
circ.swap(1, 2)

print(circ)
