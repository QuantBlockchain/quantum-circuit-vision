from braket.circuits import Circuit

circ = Circuit()

circ.ry(0, 2 * math.pi / 3)        # ① RY(2π/3) on q0
circ.ch(0, 1)                       # ② Controlled-H: ctrl=q0, target=q1
circ.ry(1, math.pi / 2)            # ③ RY(π/2) on q1
circ.cnot(1, 2)                     # ④ CNOT: ctrl=q1, target=q2
circ.ch(1, 3)                       # ⑤ Controlled-H: ctrl=q1, target=q3
circ.cnot(0, 1)                     # ⑥ CNOT: ctrl=q0, target=q1
circ.x(0)                           # ⑦ X on q0

print(circ)
