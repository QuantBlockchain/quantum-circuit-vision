[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

c = Circuit()

# Layer 1: Hadamard on q0-q3
c.h(0).h(1).h(2).h(3)

# Layer 2: CNOT q0->q4, q2->q5
c.cnot(0, 4).cnot(2, 5)

# Layer 3: CNOT q1->q4, q1->q5
c.cnot(1, 4).cnot(1, 5)

# Layer 4: CNOT q3->q4, q3->q5
c.cnot(3, 4).cnot(3, 5)

# Layer 5: Rz on q4 and q5
c.rz(4, np.pi / 4).rz(5, np.pi / 3)

# Layer 6: CNOT q4->q6, q5->q6
c.cnot(4, 6).cnot(5, 6)

# Layer 7: H on q6
c.h(6)

# Layer 8: CNOT q4->q6, q5->q6
c.cnot(4, 6).cnot(5, 6)

# Layer 9: Rz on q6
c.rz(6, np.pi / 5)

print(c)
