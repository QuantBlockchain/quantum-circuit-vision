[0m[38;5;10mfrom braket.circuits import Circuit
import math

circuit = Circuit()

# 第一层: 四个 Rz 门
circuit.rz(0, math.pi / 3)
circuit.rz(1, math.pi / 5)
circuit.rz(2, math.pi / 7)
circuit.rz(3, math.pi / 9)

# CNOT: q0 控制 q1
circuit.cnot(0, 1)

# Rz on q0
circuit.rz(0, math.pi / 4)

# CNOT: q1 控制 q2
circuit.cnot(1, 2)

# CNOT: q2 控制 q3
circuit.cnot(2, 3)

# Rz on q2
circuit.rz(2, math.pi / 6)

print(circuit)
