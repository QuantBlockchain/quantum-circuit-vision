from braket.circuits import Circuit

circuit = Circuit()

# Layer 1: Rx gates

circuit.rz(2, math.pi / 6)
circuit.rz(3, math.pi / 7)

# CNOT q0 -> q1, then Rz on q0

# CNOT q1 -> q2, then Rz on q1
circuit.cnot(1, 2)
circuit.rz(1, math.pi / 6)

# CNOT q2 -> q3, then Rz on q2 and q3
circuit.cnot(2, 3)
circuit.rz(2, math.pi / 7)
circuit.rz(3, math.pi / 8)
