circuit = Circuit()

# 初始 Rz 门

circuit.rz(3, 3.14159265359 / 9)      # Rz(π/9) on q3

circuit.cnot(0, 1)                     # CNOT: q0 -> q1
circuit.cnot(1, 2)                     # CNOT: q1 -> q2
circuit.cnot(2, 3)                     # CNOT: q2 -> q3

# 最后的 Rz 门
circuit.rz(0, 3.14159265359 / 4)      # Rz(π/4) on q0
circuit.rz(2, 3.14159265359 / 6)      # Rz(π/6) on q2

print(circuit)
