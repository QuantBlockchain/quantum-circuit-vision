[0m[38;5;10mfrom braket.circuits import Circuit

c = Circuit()

# T1: Hadamard on ancilla qubits
c.h(4)
c.h(5)

# T2: CNOT ctrl=q4, tgt=q0
c.cnot(4, 0)

# T3: CNOT ctrl=q4, tgt=q2 | ctrl=q5, tgt=q1
c.cnot(4, 2)
c.cnot(5, 1)

# T4: CNOT ctrl=q4, tgt=q3 | ctrl=q5, tgt=q6
c.cnot(4, 3)
c.cnot(5, 6)

# T5: CNOT ctrl=q0, tgt=q6 | H on q4, q5
c.cnot(0, 6)
c.h(4)
c.h(5)

# T6: CNOT ctrl=q2, tgt=q7 | ctrl=q5, tgt=q8
c.cnot(2, 7)
c.cnot(5, 8)

# T7: CNOT ctrl=q0, tgt=q7 | ctrl=q1, tgt=q8
c.cnot(0, 7)
c.cnot(1, 8)

# T8: CNOT ctrl=q3, tgt=q7 | ctrl=q1, tgt=q8
c.cnot(3, 7)
c.cnot(1, 8)

print(c)
