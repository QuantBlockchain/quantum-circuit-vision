[0m[38;5;10mfrom braket.circuits import Circuit
import math

circ = Circuit()

# 1. H on q0
circ.h(0)

# 2. Controlled-P(π/2): control=q0, target=q1
circ.cphaseshift(0, 1, math.pi / 2)

# 3. H on q1
circ.h(1)

# 4. Controlled-P(π/2): control=q0, target=q2
circ.cphaseshift(0, 2, math.pi / 2)

# 5. Controlled-P(π/4): control=q0, target=q3
circ.cphaseshift(0, 3, math.pi / 4)

# 6. Controlled-P(π/2): control=q1, target=q3
circ.cphaseshift(1, 3, math.pi / 2)

# 7. H on q1 (second H)
circ.h(1)

# 8. Controlled-P(-π/2): control=q1, target=q0
circ.cphaseshift(1, 0, -math.pi / 2)

# 9. H on q0 (rightmost)
circ.h(0)

print(circ)
