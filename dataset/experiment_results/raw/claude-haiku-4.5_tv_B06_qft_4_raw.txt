[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

# 创建 4-qubit 量子电路
circuit = Circuit()

# 第一层：H 门在 q0
circuit.h(0)

# 第一组受控 P 门（控制：q0）
circuit.cphaseshift(0, 1, np.pi / 2)
circuit.cphaseshift(0, 2, np.pi / 4)
circuit.cphaseshift(0, 3, np.pi / 8)

# 第二层：H 门在 q1
circuit.h(1)

# 第二组受控 P 门（控制：q1）
circuit.cphaseshift(1, 2, np.pi / 2)
circuit.cphaseshift(1, 3, np.pi / 4)

# 第三层：H 门在 q2
circuit.h(2)

# 第三组受控 P 门（控制：q2）
circuit.cphaseshift(2, 3, np.pi / 2)

# 第四层：H 门在 q3
circuit.h(3)

# 测量所有 qubit
circuit.measure_all()

# 打印电路
print(circuit)
