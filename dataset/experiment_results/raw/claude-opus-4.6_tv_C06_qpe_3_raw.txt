[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circ = Circuit()

# 初始化
circ.h(0).h(1).x(2)

# 受控相位门 (QPE 核心)
circ.cphaseshift(0, 2, np.pi / 2)

# 逆 QFT
circ.h(0)
circ.cphaseshift(1, 2, np.pi / 4)
circ.cphaseshift(0, 1, -np.pi / 2)
circ.h(1)
circ.swap(0, 1)

print(circ)
