[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circuit = Circuit()

# Q0
circuit.h(0)
circuit.cphaseshift(0, 1, np.pi / 2)

# Q1
circuit.h(1)
circuit.cphaseshift(0, 2, np.pi / 4)
circuit.cphaseshift(1, 2, np.pi / 8)

# Q2
circuit.h(2)
circuit.cphaseshift(0, 3, np.pi / 16)
circuit.cphaseshift(1, 3, np.pi / 8)
circuit.cphaseshift(2, 3, np.pi / 4)

# Q3
circuit.h(3)
circuit.cphaseshift(0, 4, np.pi / 2)
circuit.cphaseshift(1, 4, np.pi / 4)
circuit.cphaseshift(2, 4, np.pi / 8)
circuit.cphaseshift(3, 4, np.pi / 2)

# Q4
circuit.h(4)

print(circuit)
