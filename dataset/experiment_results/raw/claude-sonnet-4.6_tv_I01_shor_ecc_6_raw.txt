[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circ = Circuit()

# Step 1: H on q0, q1, q2; X on q5
circ.h(0).h(1).h(2).x(5)

# Step 2: CNOT q2 -> q3
circ.cnot(2, 3)

# Step 3: CNOT q2 -> q4, CNOT q1 -> q3
circ.cnot(2, 4).cnot(1, 3)

# Step 4: CNOT q1 -> q4, CNOT q2 -> q3
circ.cnot(1, 4).cnot(2, 3)

# Step 5: CNOT q0 -> q5, CNOT q1 -> q5, CNOT q0 -> q3
circ.cnot(0, 5).cnot(1, 5).cnot(0, 3)

# Step 6: CNOT q0 -> q5, CNOT q1 -> q5
circ.cnot(0, 5).cnot(1, 5)

# Step 7: H on q0
circ.h(0)

# Step 8: CP(-π/2) q0 -> q1
circ.cphaseshift(0, 1, -np.pi / 2)

# Step 9: CP(-π/4) q0 -> q2, H on q1
circ.cphaseshift(0, 2, -np.pi / 4).h(1)

# Step 10: CP(-π/2) q1 -> q2
circ.cphaseshift(1, 2, -np.pi / 2)

# Step 11: H on q2
circ.h(2)

# Step 12: SWAP q0 <-> q2
circ.swap(0, 2)

print(circ)
