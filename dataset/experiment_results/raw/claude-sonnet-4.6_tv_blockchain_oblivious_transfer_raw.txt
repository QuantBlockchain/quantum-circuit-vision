[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circ = Circuit()

# Step 1: H on q0, q1
circ.h(0).h(1)

# Step 2: CNOT q0 -> q2
circ.cnot(0, 2)

# Step 3: CNOT q1 -> q3
circ.cnot(1, 3)

# Step 4: H on q0, q1, q2, q3
circ.h(0).h(1).h(2).h(3)

# Step 5: CNOT q2 -> q4
circ.cnot(2, 4)

# Step 6: CNOT q3 -> q4
circ.cnot(3, 4)

# Step 7: Rz(pi/3) on q4
circ.rz(4, np.pi / 3)

# Step 8: CNOT q2 -> q4
circ.cnot(2, 4)

# Step 9: CNOT q3 -> q4
circ.cnot(3, 4)

print(circ)
