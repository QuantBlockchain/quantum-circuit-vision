import math

circuit = Circuit()

# First layer: H gates on all qubits
for i in range(5):
    circuit.h(i)

# CNOT chain
circuit.cnot(0, 1)
circuit.cnot(1, 2)
circuit.cnot(2, 3)
circuit.cnot(3, 4)

# Rz gates with specific angles
circuit.rz(0, math.pi / 3)
circuit.rz(1, math.pi / 5)
circuit.rz(2, math.pi / 7)
circuit.rz(3, math.pi / 11)
circuit.rz(4, math.pi / 13)

# Last layer: H gates on all qubits
for i in range(5):
    circuit.h(i)

print(circuit)
