from braket.circuits import Circuit

circ = Circuit()

# 1. H on q0
circ.h(0)

# 2. CP(π/2): control=q0, target=q1
circ.cphaseshift(0, 1, math.pi / 2)

# 3. H on q1
circ.h(1)

# 4. CP(π/2): control=q2, target=q1
circ.cphaseshift(2, 1, math.pi / 2)

# 5. CP(π/2): control=q0, target=q1
circ.cphaseshift(0, 1, math.pi / 2)

# 6. CP(π/4): control=q3, target=q0
circ.cphaseshift(3, 0, math.pi / 4)

# 7. CP(π/2): control=q3, target=q1
circ.cphaseshift(3, 1, math.pi / 2)

# 8. H on q1
circ.h(1)

# 9. CP(-π/2): control=q1, target=q0
circ.cphaseshift(1, 0, -math.pi / 2)

# 10. H on q0
circ.h(0)

print(circ)
