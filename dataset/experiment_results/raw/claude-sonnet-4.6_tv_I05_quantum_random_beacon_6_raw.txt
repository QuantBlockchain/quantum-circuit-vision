[0m[38;5;10mfrom braket.circuits import Circuit
import math

circ = Circuit()

# 第1列：单量子比特门
circ.h(0)
circ.rz(1, math.pi / 3)
circ.rz(3, math.pi / 5)
circ.rz(5, math.pi / 7)

# 第2列：CNOT q0 -> q2
circ.cnot(0, 2)

# 第3列：CNOT q0 -> q4
circ.cnot(0, 4)

# 第4列：CNOT q0->q1, q2->q3, q4->q5
circ.cnot(0, 1)
circ.cnot(2, 3)
circ.cnot(4, 5)

# 第5列：H on q0, q2, q4
circ.h(0)
circ.h(2)
circ.h(4)

print(circ)
