[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circ = Circuit()

# QFT on q0
circ.h(0)
circ.cphaseshift(1, 0, np.pi / 2)
circ.cphaseshift(2, 0, np.pi / 4)
circ.cphaseshift(3, 0, np.pi / 8)

# QFT on q1
circ.h(1)
circ.cphaseshift(2, 1, np.pi / 2)
circ.cphaseshift(3, 1, np.pi / 4)

# QFT on q2
circ.h(2)
circ.cphaseshift(3, 2, np.pi / 2)

# QFT on q3
circ.h(3)

# Bit-reversal swaps
circ.swap(0, 3)
circ.swap(1, 2)

print(circ)
