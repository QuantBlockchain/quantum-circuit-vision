[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circuit = Circuit()

# q0: H, then controlled-P gates (control=q1,q2,q3,q4)
circuit.h(0)
circuit.cphaseshift(1, 0, np.pi/2)
circuit.cphaseshift(2, 0, np.pi/4)
circuit.cphaseshift(3, 0, np.pi/8)
circuit.cphaseshift(4, 0, np.pi/16)

# q1: H, then controlled-P gates (control=q2,q3,q4)
circuit.h(1)
circuit.cphaseshift(2, 1, np.pi/2)
circuit.cphaseshift(3, 1, np.pi/4)
circuit.cphaseshift(4, 1, np.pi/8)

# q2: H, then controlled-P gates (control=q3,q4)
circuit.h(2)
circuit.cphaseshift(3, 2, np.pi/2)
circuit.cphaseshift(4, 2, np.pi/4)

# q3: H, then controlled-P gate (control=q4)
circuit.h(3)
circuit.cphaseshift(4, 3, np.pi/2)

# q4: H
circuit.h(4)

# Swap q0 <-> q4, q1 <-> q3
circuit.swap(0, 4)
circuit.swap(1, 3)

print(circuit)
