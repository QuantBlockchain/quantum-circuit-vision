# 创建量子电路
circuit = Circuit()

# q0: H 门
circuit.h(0)

# 受控 P 门：q0 控制，q1 目标，角度 π/2
circuit.cphaseshift(0, 1, angle=3.14159265359/2)

# q1: H 门
circuit.h(1)

# 受控 NOT 门（CNOT/CX）：q1 控制，q0 目标
circuit.cnot(1, 0)

# 测量所有 qubit
circuit.measure_all()

# 打印电路
print(circuit)
