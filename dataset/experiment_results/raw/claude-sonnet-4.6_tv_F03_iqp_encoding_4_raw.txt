import numpy as np

circ = Circuit()

# Step 1: H on all qubits
circ.h(0).h(1).h(2).h(3)

# Step 2-4: CNOT(q0->q1), Rz(π/4) on q1, CNOT(q0->q1)

circ.rz(1, np.pi / 4)
circ.cnot(0, 1)

# Step 5-8: CNOT(q1->q2), H on q0, Rz(π/4) on q2, CNOT(q1->q2)
circ.cnot(1, 2)
circ.h(0)
circ.rz(2, np.pi / 4)
circ.cnot(1, 2)

# Step 9-12: CNOT(q2->q3), H on q1, Rz(π/4) on q3, CNOT(q2->q3)
circ.cnot(2, 3)
circ.h(1)
circ.rz(3, np.pi / 4)
circ.cnot(2, 3)

# Step 13: H on q2, q3
circ.h(2).h(3)

print(circ)
