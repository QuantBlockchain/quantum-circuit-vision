[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circ = Circuit()

# q0: H
circ.h(0)

# CP(π/2): control=q0, target=q1
circ.cphaseshift(0, 1, np.pi / 2)

# CP(π/4): control=q0, target=q2
circ.cphaseshift(0, 2, np.pi / 4)

# q1: H
circ.h(1)

# CP(π/2): control=q1, target=q2
circ.cphaseshift(1, 2, np.pi / 2)

# q2: H
circ.h(2)

print(circ)
