[0m[38;5;10mimport math
from braket.circuits import Circuit

circuit = Circuit()

# Step 1: H gates
circuit.h([0, 2, 4, 6])

# Step 2-5: First round of CNOTs
circuit.cnot(0, 1)
circuit.cnot(2, 3)
circuit.cnot(4, 5)
circuit.cnot(6, 7)

# Step 6-9: Second round of CNOTs (cross-pair entanglement)
circuit.cnot(1, 2)
circuit.cnot(3, 4)
circuit.cnot(5, 6)
circuit.cnot(7, 0)

# Step 10: Rz(π/4) gates
circuit.rz([0, 2, 4, 6], math.pi / 4)

# Step 11: H gates
circuit.h([0, 2, 4, 6])

print(circuit)
