# 第3步：Hadamard 门
circuit.h(0)
circuit.h(1)
circuit.x(2)
circuit.h(3)

# 第4步：X 门

circuit.x(1)
circuit.x(2)

# 第5-7步：H, X, H 在 q2
circuit.h(2)
circuit.x(2)
circuit.h(2)

# 第8步：Toffoli 门 (q0, q1 控制，q2 目标)
circuit.ccnot(0, 1, 2)

# 第9-11步：最后的门
circuit.x(0)
circuit.x(1)
circuit.h(2)
circuit.x(2)
circuit.h(2)

# 打印电路
print(circuit)
