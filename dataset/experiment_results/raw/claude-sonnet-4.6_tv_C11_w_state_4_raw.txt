[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circ = Circuit()

# 1. RY(2π/3) on q0
circ.ry(0, 2 * np.pi / 3)

# 2. Controlled-H: control q0, target q1
circ.ch(0, 1)

# 3. RY(π/2) on q1
circ.ry(1, np.pi / 2)

# 4. CNOT: control q1, target q2
circ.cnot(1, 2)

# 5. H on q3
circ.h(3)

# 6. CNOT: control q1, target q3
circ.cnot(1, 3)

# 7. CNOT: control q0, target q1
circ.cnot(0, 1)

# 8. X on q0
circ.x(0)

print(circ)
