[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# Initial H gates on counting register + X on target
circ.h(0).h(1).x(3)

# Controlled phase gates
circ.cphaseshift(0, 2, math.pi / 2)
circ.cphaseshift(0, 3, math.pi / 4)

circ.h(0)

circ.cphaseshift(1, 2, math.pi / 4)
circ.cphaseshift(1, 3, math.pi / 8)

# Inverse QFT on counting register
circ.cphaseshift(0, 1, -math.pi / 2)
circ.h(1)
circ.swap(0, 1)

print(circ)
