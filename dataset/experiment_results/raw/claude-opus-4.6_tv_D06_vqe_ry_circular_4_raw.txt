[0m[38;5;10mimport math
from braket.circuits import Circuit

circuit = Circuit()

# Layer 1: RY rotations
circuit.ry(0, math.pi / 3)
circuit.ry(1, math.pi / 4)
circuit.ry(2, math.pi / 5)
circuit.ry(3, math.pi / 6)

# Layer 2: Circular CNOT entanglement
circuit.cnot(0, 1)
circuit.cnot(1, 2)
circuit.cnot(2, 3)

# RY on q1 and circular CNOT q3->q0
circuit.ry(1, math.pi / 6)
circuit.cnot(3, 0)

# Layer 3: Final RY rotations
circuit.ry(0, math.pi / 5)
circuit.ry(2, math.pi / 7)
circuit.ry(3, math.pi / 8)

print(circuit)
