from braket.circuits import Circuit

circuit = Circuit()

# Initial superposition
circuit.h(0).h(1).h(2).h(3)

# === Layer 1: Problem Hamiltonian (gamma_1, Rz angles = pi/2) ===
circuit.cnot(0, 1).rz(1, math.pi / 2).cnot(0, 1)
circuit.cnot(1, 2).rz(2, math.pi / 2).cnot(1, 2)
circuit.cnot(2, 3).rz(3, math.pi / 2).cnot(2, 3)

# === Layer 1: Mixer Hamiltonian (beta_1, Rx angles = pi/4) ===
circuit.cnot(0, 1).rz(0, math.pi / 2).cnot(0, 1)
circuit.rx(1, math.pi / 4)
circuit.cnot(1, 2).rx(2, math.pi / 4).cnot(1, 2)
circuit.cnot(2, 3).rx(3, math.pi / 4).cnot(2, 3)
circuit.rx(0, math.pi / 4)

# === Layer 2: Problem Hamiltonian (gamma_2, Rz angles = pi/3) ===
circuit.cnot(0, 1).rz(1, math.pi / 3).cnot(0, 1)
circuit.cnot(1, 2).rz(2, math.pi / 3).cnot(1, 2)
circuit.cnot(2, 3).rz(3, math.pi / 3).cnot(2, 3)

# === Layer 2: Mixer Hamiltonian (beta_2, Rx angles = pi/5) ===

circuit.cnot(2, 3).rx(3, math.pi / 5).cnot(2, 3)
circuit.rx(0, math.pi / 5)

circuit.cnot(3, 0).rx(0, math.pi / 5).rx(3, math.pi / 5)
