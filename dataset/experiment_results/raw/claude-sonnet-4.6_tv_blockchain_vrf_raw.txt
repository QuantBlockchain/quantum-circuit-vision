[0m[38;5;10mfrom braket.circuits import Circuit
import math

c = Circuit()

# Layer 1: Hadamard
c.h(0).h(3).h(4).h(5)

# CNOT gates (blue section)
c.cnot(0, 1)
c.cnot(0, 2)
c.cnot(3, 1)
c.cnot(4, 0)
c.cnot(3, 2)
c.cnot(5, 0)

# Rz + controlled operations (purple section)
c.rz(4, math.pi / 6)
c.rz(3, math.pi / 4)
c.h(4)
c.rz(5, math.pi / 8)

# Controlled-Rz gates
c.crz(1, 3, math.pi / 4)   # control q1, target q3
c.crz(0, 3, math.pi / 4)   # control q0, target q3
c.crz(2, 5, math.pi / 8)   # control q2, target q5

# Final Hadamard
c.h(3).h(5)

print(c)
