[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# Layer 1: Rz rotations
circ.rz(0, math.pi / 3)
circ.rz(1, math.pi / 5)
circ.rz(2, math.pi / 7)
circ.rz(3, math.pi / 9)

# Layer 2: CNOT q0 -> q1
circ.cnot(0, 1)

# Layer 3: Rz on q0, then CNOT q1->q2 and q1->q3
circ.rz(0, math.pi / 4)
circ.cnot(1, 2)
circ.cnot(1, 3)

# Layer 4: CNOT q2->q3, Rz on q2
circ.cnot(2, 3)
circ.rz(2, math.pi / 6)

print(circ)
