[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circ = Circuit()

# 第1层
circ.h(0).h(1).x(3)

# 第2层：CNOT, 控制q1 → 目标q2
circ.cnot(1, 2)

# 第3层：Toffoli(q0,q1→q2) 和 CNOT(q1→q3)
circ.ccnot(0, 1, 2)
circ.cnot(1, 3)

# 第4层：CNOT, 控制q0 → 目标q3
circ.cnot(0, 3)

# 第5层：H on q0
circ.h(0)

# 第6层：受控相位 P(-π/2), 控制q0 → 目标q1
circ.cphaseshift(0, 1, -np.pi / 2)

# 第7层：H on q1
circ.h(1)

print(circ)
