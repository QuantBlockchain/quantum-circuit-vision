[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# Initial H layer
circ.h(range(4))

# ZZ(pi/4) on q0-q1: CNOT - Rz - CNOT
circ.cnot(0, 1)
circ.rz(1, math.pi / 4)
circ.cnot(0, 1)

# ZZ(pi/4) on q1-q2: CNOT - Rz - CNOT
circ.cnot(1, 2)
circ.h(0)
circ.rz(2, math.pi / 4)
circ.cnot(2, 3)
circ.cnot(1, 2)
circ.h(1)

# ZZ(pi/4) on q2-q3: CNOT - Rz - CNOT
circ.cnot(2, 3)
circ.rz(3, math.pi / 4)
circ.cnot(2, 3)

# Final H layer
circ.h(2)
circ.h(3)

print(circ)
