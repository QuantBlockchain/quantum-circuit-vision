[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# QFT on q0
circ.h(0)
circ.cphaseshift(0, 1, math.pi / 2)
circ.cphaseshift(0, 2, math.pi / 4)
circ.cphaseshift(0, 3, math.pi / 8)
circ.cphaseshift(0, 4, math.pi / 16)

# QFT on q1
circ.h(1)
circ.cphaseshift(1, 2, math.pi / 2)
circ.cphaseshift(1, 3, math.pi / 4)
circ.cphaseshift(1, 4, math.pi / 8)

# QFT on q2
circ.h(2)
circ.cphaseshift(2, 3, math.pi / 2)
circ.cphaseshift(2, 4, math.pi / 4)

# QFT on q3
circ.h(3)
circ.cphaseshift(3, 4, math.pi / 2)

# QFT on q4
circ.h(4)

# Bit-reversal swaps
circ.swap(0, 4)
circ.swap(1, 3)

print(circ)
