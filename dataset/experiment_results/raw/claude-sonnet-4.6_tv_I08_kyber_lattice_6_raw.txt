[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# Layer 1: Hadamard on q0, q1, q2
circ.h(0).h(1).h(2)

# Layer 2-4: CNOT q0->q3, q1->q4, q2->q5
circ.cnot(0, 3)
circ.cnot(1, 4)
circ.cnot(2, 5)

# Layer 5: RZ rotations
circ.rz(3, math.pi / 4)
circ.rz(4, math.pi / 3)
circ.rz(5, math.pi / 5)

# Layer 6: CNOT q3->q4
circ.cnot(3, 4)

# Layer 7: CNOT q4->q5
circ.cnot(4, 5)

# Layer 8: CNOT q4->q3, CNOT q5->q3, CNOT q5->q4
circ.cnot(4, 3)
circ.cnot(5, 3)
circ.cnot(5, 4)

# Layer 9: Hadamard on q3, q4, q5
circ.h(3).h(4).h(5)

print(circ)
