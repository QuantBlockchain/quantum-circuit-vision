[0m[38;5;10mfrom braket.circuits import Circuit
import math

c = Circuit()

# Layer 1: Hadamard on all qubits
c.h(0).h(1).h(2).h(3)

# CNOT: q0 -> q1, then RZ(π/2) on q1, then CNOT: q0 -> q1
c.cnot(0, 1)
c.rz(1, math.pi / 2)
c.cnot(0, 1)

# CNOT: q1 -> q2, then RZ(π/2) on q2, then CNOT: q1 -> q2
c.cnot(1, 2)
c.rz(2, math.pi / 2)
c.cnot(1, 2)

# CNOT: q1 -> q3, then RZ(π/2) on q3, then CNOT: q1 -> q3
c.cnot(1, 3)
c.rz(3, math.pi / 2)
c.cnot(1, 3)

# RX(π/4) on q1
c.rx(1, math.pi / 4)

# CNOT: q2 -> q3
c.cnot(2, 3)

# CNOT: q3 -> q0, then RZ(π/2) on q0, then CNOT: q3 -> q0
c.cnot(3, 0)
c.rz(0, math.pi / 2)
c.rx(2, math.pi / 4)
c.cnot(3, 0)

# Final RX(π/4) on q0, q3
c.rx(0, math.pi / 4)
c.rx(3, math.pi / 4)

print(c)
