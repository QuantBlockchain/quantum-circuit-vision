c = Circuit()

# Step 1: H on q2
c.h(2)
# Step 2: CNOT q1 -> q2
c.cnot(1, 2)
# Step 3: T† on q2
c.ti(2)
# Step 4: CNOT q0 -> q2
c.cnot(0, 2)
# Step 5: T on q2
c.t(2)
# Step 6: CNOT q1 -> q2
c.cnot(1, 2)
# Step 7: T† on q2
c.ti(2)
# Step 8: CNOT q0 -> q2
c.cnot(0, 2)
# Step 9: T on q1, T† on q2 (parallel)
c.t(1)
c.ti(2)
# Step 10: CNOT q0 -> q1
c.cnot(0, 1)
# Step 11: T on q0, T† on q1
c.t(0)
c.ti(1)
# Step 12: CNOT q0 -> q1
c.cnot(0, 1)
# Step 13: H on q2
c.h(2)

print(c)
