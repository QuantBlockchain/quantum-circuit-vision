[1mpython
[0m[38;5;10mfrom braket.circuits import Circuit, Gate, Instruction
import numpy as np

c = Circuit()
c.h(1)
c.ry(2, np.pi / 3)
c.cphaseshift(1, 2, np.pi / 2)
c.h(1)

# Controlled-RY(π/4): control=q1, target=q0
cry = Gate.Ry(angle=np.pi / 4).controlled(1)
c.add_instruction(Instruction(cry, [1, 0]))

print(c)
