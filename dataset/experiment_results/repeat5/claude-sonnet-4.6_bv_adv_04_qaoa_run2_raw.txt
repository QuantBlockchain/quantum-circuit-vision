[0m[38;5;10mfrom braket.circuits import Circuit
import math

circ = Circuit()

# Initial Hadamard layer
circ.h(0).h(1).h(2).h(3)

# CNOT: q0 -> q1, then RZ(pi/2) on q1, then CNOT: q0 -> q1
circ.cnot(0, 1)
circ.rz(1, math.pi / 2)
circ.cnot(0, 1)

# CNOT: q1 -> q2, then RZ(pi/2) on q2, then CNOT: q1 -> q2
circ.cnot(1, 2)
circ.rz(2, math.pi / 2)
circ.cnot(1, 2)

# CNOT: q1 -> q3, then RZ(pi/2) on q3, then CNOT: q1 -> q3
circ.cnot(1, 3)
circ.rz(3, math.pi / 2)
circ.cnot(1, 3)

# RX(pi/4) on q1
circ.rx(1, math.pi / 4)

# CNOT: q2 -> q3, then CNOT: q2 -> q3 (already done above for q1->q3)
# From diagram: q2 control -> q3 target
circ.cnot(2, 3)

# RX(pi/4) on q2
circ.rx(2, math.pi / 4)

# CNOT: q3 -> q0
circ.cnot(3, 0)
circ.rz(0, math.pi / 2)
circ.cnot(3, 0)

# RX(pi/4) on q0
circ.rx(0, math.pi / 4)

# RX(pi/4) on q3
circ.rx(3, math.pi / 4)
