[0m[38;5;10mimport math
from braket.circuits import Circuit

circuit = Circuit()

# 第一层 RY 旋转
circuit.ry(0, math.pi / 4)
circuit.ry(1, math.pi / 3)
circuit.ry(2, math.pi / 6)
circuit.ry(3, math.pi / 5)

# 级联 CNOT 纠缠层
circuit.cnot(0, 1)
circuit.cnot(1, 2)
circuit.cnot(2, 3)

# 第二层 RY 旋转
circuit.ry(0, math.pi / 5)
circuit.ry(1, math.pi / 6)
circuit.ry(2, math.pi / 3)
circuit.ry(3, math.pi / 4)

print(circuit)
