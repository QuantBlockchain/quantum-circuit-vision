[0m[38;5;10mfrom braket.circuits import Circuit

circuit = Circuit()

# H gate on q0
circuit.h(0)

# Controlled Phase gates
circuit.cphaseshift(0, 1, 3.14159265359 / 2)  # CPhase(π/2) on q0, q1
circuit.cphaseshift(0, 2, 3.14159265359 / 4)  # CPhase(π/4) on q0, q2

# H gate on q1
circuit.h(1)

# Controlled Phase gate
circuit.cphaseshift(1, 2, 3.14159265359 / 2)  # CPhase(π/2) on q1, q2

# H gate on q2
circuit.h(2)

# SWAP gate between q0 and q2
circuit.swap(0, 2)

print(circuit)
