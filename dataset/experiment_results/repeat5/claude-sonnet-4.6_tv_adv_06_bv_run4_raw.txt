circ = Circuit()

# Step 1: 初始化
circ.x(4)

# Step 2: q2 额外的 H（q2:H 在第一列之后紧跟一个 H，相当于 HH=I，可保留以忠实还原电路）

circ.h(0)
circ.cnot(3, 4)
circ.h([1, 3])

print(circ)
