[0m[38;5;10mfrom braket.circuits import Circuit
import math

circ = Circuit()

# Initial Hadamard layer
circ.h(0).h(1).h(2).h(3)

# CNOT(0,1), RZ(pi/2) on q1, CNOT(0,1)
circ.cnot(0, 1).rz(1, math.pi/2).cnot(0, 1)

# CNOT(1,2), RZ(pi/2) on q2, CNOT(1,2)
circ.cnot(1, 2).rz(2, math.pi/2).cnot(1, 2)

# CNOT(2,3), RZ(pi/2) on q3, CNOT(2,3)
circ.cnot(2, 3).rz(3, math.pi/2).cnot(2, 3)

# RX(pi/4) on q1
circ.rx(1, math.pi/4)

# CNOT(2,3) again (mixer layer)
circ.cnot(2, 3)

# RX(pi/4) on q2
circ.rx(2, math.pi/4)

# CNOT(0,1) for mixer
circ.cnot(0, 1)

# RZ(pi/2) on q0, CNOT(0,1)
circ.rz(0, math.pi/2).cnot(0, 1)

# RX(pi/4) on q0
circ.rx(0, math.pi/4)

# CNOT(3,0), CNOT(3,1)
circ.cnot(3, 0).cnot(3, 1)

# RX(pi/4) on q3
circ.rx(3, math.pi/4)

print(circ)
