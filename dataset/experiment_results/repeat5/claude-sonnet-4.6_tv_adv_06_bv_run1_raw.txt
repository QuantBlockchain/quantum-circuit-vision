[0m[38;5;10mfrom braket.circuits import Circuit

circ = Circuit()

# Step 1: X on ancilla q4
circ.x(4)

# Step 2: H on all qubits
for q in range(5):
    circ.h(q)

# Step 3: Oracle CNOTs (hidden string s=1011, bits q0,q1,q3)
circ.cnot(0, 4)
circ.cnot(1, 4)
circ.cnot(3, 4)

# Step 4: H on input qubits (q0, q1, q3; q2 skipped as its two H gates cancel)
circ.h(0)
circ.h(1)
circ.h(3)
