circ = Circuit()

# 初始化：对输入 qubits 施加 H，辅助 qubit q4 施加 X+H

circ.x(4)

# Oracle：CNOT gates（隐藏字符串 s=101 对应 q0, q1, q3）
circ.cnot(0, 4)
circ.cnot(1, 4)
circ.cnot(3, 4)

# 末尾 H 门（BV 算法测量前）
circ.h(0)
circ.h(1)
circ.h(3)

print(circ)
