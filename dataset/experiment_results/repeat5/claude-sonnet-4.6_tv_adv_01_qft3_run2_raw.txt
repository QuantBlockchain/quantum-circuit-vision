[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circuit = Circuit()

# Step 1: H on q0
circuit.h(0)

# Step 2: Controlled-Phase on q0->q1, angle π/2
circuit.cphaseshift(0, 1, np.pi / 2)

# Step 3: Controlled-Phase on q0->q2, angle π/4
circuit.cphaseshift(0, 2, np.pi / 4)

# Step 4: H on q1
circuit.h(1)

# Step 5: Controlled-Phase on q1->q2, angle π/2
circuit.cphaseshift(1, 2, np.pi / 2)

# Step 6: H on q2
circuit.h(2)

# Step 7: SWAP q0 <-> q2
circuit.swap(0, 2)

print(circuit)
