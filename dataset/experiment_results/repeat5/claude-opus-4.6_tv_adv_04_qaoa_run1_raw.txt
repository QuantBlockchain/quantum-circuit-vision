[0m[38;5;10mimport math
from braket.circuits import Circuit

gamma = math.pi / 2
beta = math.pi / 4

circ = Circuit()

# Initial superposition
circ.h(range(4))

# Cost layer: ZZ interactions via CNOT-Rz-CNOT
for ctrl, tgt in [(0, 1), (1, 2), (2, 3)]:
    circ.cnot(ctrl, tgt).rz(tgt, gamma).cnot(ctrl, tgt)

# Mixer layer: Rx on q1
circ.rx(1, beta)

# Cost layer: ZZ(q2, q0)
circ.cnot(2, 0).rz(0, gamma).cnot(2, 0)

# Mixer layer: Rx on q2
circ.rx(2, beta)

# Cost layer: ZZ(q3, q0)
circ.cnot(3, 0)

# Mixer layer: Rx on q0, q3
circ.rx(0, beta)
circ.rx(3, beta)

print(circ)
