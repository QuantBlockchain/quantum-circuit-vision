[0m[38;5;10mfrom braket.circuits import Circuit
import math

circuit = Circuit()

# Initial Hadamard gates on all qubits
circuit.h(0)
circuit.h(1)
circuit.h(2)
circuit.h(3)

# First layer of CNOTs and RZ gates
circuit.cnot(0, 1)
circuit.rz(1, math.pi / 2)
circuit.cnot(1, 0)

circuit.cnot(1, 2)
circuit.rz(2, math.pi / 2)
circuit.cnot(2, 1)
circuit.rx(1, math.pi / 4)

circuit.cnot(2, 3)
circuit.rz(3, math.pi / 2)
circuit.cnot(3, 2)

# Second layer of CNOTs and RZ gates
circuit.cnot(0, 3)
circuit.rz(0, math.pi / 2)
circuit.cnot(3, 0)

# Final RX gates
circuit.rx(2, math.pi / 4)
circuit.rx(0, math.pi / 4)
circuit.rx(3, math.pi / 4)

print(circuit)
