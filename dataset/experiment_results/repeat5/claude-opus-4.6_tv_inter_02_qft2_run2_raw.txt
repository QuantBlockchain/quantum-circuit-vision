[0m[38;5;10mimport numpy as np
from braket.circuits import Circuit

circuit = Circuit()

# H on q0
circuit.h(0)

# Controlled-P(π/2): control=q0, target=q1
circuit.cphaseshift(0, 1, np.pi / 2)

# H on q1
circuit.h(1)

# SWAP q0 and q1
circuit.swap(0, 1)

print(circuit)
