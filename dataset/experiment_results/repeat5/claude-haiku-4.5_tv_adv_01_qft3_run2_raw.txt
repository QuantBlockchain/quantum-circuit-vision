[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circuit = Circuit()

# H gate on q0
circuit.h(0)

# Controlled-P(π/2) with control on q0, target on q1
circuit.cphaseshift(0, 1, np.pi / 2)

# Controlled-P(π/4) with control on q0, target on q2
circuit.cphaseshift(0, 2, np.pi / 4)

# H gate on q1
circuit.h(1)

# Controlled-P(π/2) with control on q1, target on q2
circuit.cphaseshift(1, 2, np.pi / 2)

# H gate on q2
circuit.h(2)

# SWAP gate between q0 and q2
circuit.swap(0, 2)

print(circuit)
