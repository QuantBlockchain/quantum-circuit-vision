[0m[38;5;10mimport math
from braket.circuits import Circuit

c = Circuit()

# 初始叠加
c.h(range(4))

# ZZ interactions (cost layer)
# q0-q1
c.cnot(0, 1).rz(1, math.pi / 2).cnot(0, 1)
# q1-q2
c.cnot(1, 2).rz(2, math.pi / 2).cnot(1, 2)
# q2-q3
c.cnot(2, 3).rz(3, math.pi / 2).cnot(2, 3)
# q2-q0
c.cnot(2, 0).rz(0, math.pi / 2).cnot(2, 0)

# Mixer layer
c.rx(1, math.pi / 4)
c.rx(2, math.pi / 4)

# q3-q0 ZZ + remaining mixers
c.cnot(3, 0)
c.rx(0, math.pi / 4)
c.rx(3, math.pi / 4)

print(c)

电路结构说明：[0m[0m
- **Cost layer**：4 对 ZZ 交互（q₀-q₁, q₁-q₂, q₂-q₃, q₂-q₀），每对用 CNOT-Rz(π/2)-CNOT 分解实现。[0m[0m
- **Mixer layer**：对各 qubit 施加 Rx(π/4) 旋转。[0m[0m
