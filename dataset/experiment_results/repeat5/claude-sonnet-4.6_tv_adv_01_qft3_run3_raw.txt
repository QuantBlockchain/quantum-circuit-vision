[0m[38;5;10mfrom braket.circuits import Circuit
import numpy as np

circ = Circuit()

# Step 1: H on q0
circ.h(0)

# Step 2: Controlled-Phase(π/2), control=q0, target=q1
circ.cphaseshift(0, 1, np.pi / 2)

# Step 3: Controlled-Phase(π/4), control=q0, target=q2
circ.cphaseshift(0, 2, np.pi / 4)

# Step 4: H on q1
circ.h(1)

# Step 5: Controlled-Phase(π/2), control=q1, target=q2
circ.cphaseshift(1, 2, np.pi / 2)

# Step 6: H on q2
circ.h(2)

# Step 7: SWAP q0 and q2
circ.swap(0, 2)

print(circ)
