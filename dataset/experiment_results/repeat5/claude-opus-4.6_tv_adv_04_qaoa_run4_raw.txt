[0m[38;5;10mimport math
from braket.circuits import Circuit

circ = Circuit()

# 初始叠加
circ.h(range(4))

# ZZ interactions (problem unitary)
# ZZ(q0, q1)
circ.cnot(0, 1).rz(1, math.pi / 2).cnot(0, 1)
# ZZ(q1, q2)
circ.cnot(1, 2).rz(2, math.pi / 2).cnot(1, 2)
# ZZ(q2, q3)
circ.cnot(2, 3).rz(3, math.pi / 2).cnot(2, 3)

# Rx (mixer unitary - partial)
circ.rx(1, math.pi / 4)

# ZZ(q2, q0)
circ.cnot(2, 0).rz(0, math.pi / 2).cnot(2, 0)

# Rx (mixer unitary - partial)
circ.rx(2, math.pi / 4)

# ZZ(q3, q0)
circ.cnot(3, 0)

# Rx (mixer unitary - remaining)
circ.rx(0, math.pi / 4)
circ.rx(3, math.pi / 4)

print(circ)
