circuit = Circuit()

# CX(q0, q1)
circuit.cnot(0, 1)

# Rz(π/2) on q1

# CX(q1, q2)
circuit.cnot(1, 2)

# Rz(π/2) on q2
circuit.rz(2, math.pi / 2)

# CX(q2, q3)
circuit.cnot(2, 3)

# Rz(π/2) on q3
circuit.rz(3, math.pi / 2)

# Rx(π/4) on q1
circuit.rx(1, math.pi / 4)

# Rx(π/4) on q2
circuit.rx(2, math.pi / 4)

# CX(q0, q3)
circuit.cnot(0, 3)

# Rz(π/2) on q0
circuit.rz(0, math.pi / 2)

# CX(q0, q3)
circuit.cnot(0, 3)

# Rx(π/4) on q0

# Rx(π/4) on q3

print(circuit)
